# Carregamento e conversão para formato long
df = pd.read_excel("base_ceara.xlsx")
df_long = df.iloc[1:].melt(id_vars=["Unnamed: 0"], var_name="mes_ano", value_name="y")
df_long = df_long.rename(columns={"Unnamed: 0": "unique_id"})

# Tratamento de datas e valores numéricos
meses_pt = {
    "janeiro": "01", "fevereiro": "02", "março": "03", "abril": "04",
    "maio": "05", "junho": "06", "julho": "07", "agosto": "08",
    "setembro": "09", "outubro": "10", "novembro": "11", "dezembro": "12"
}

def converte_data(texto):
    mes, ano = texto.strip().split(" ")
    return pd.to_datetime(f"{ano}-{meses_pt[mes.lower()]}-01")

df_long["ds"] = df_long["mes_ano"].apply(converte_data)
df_long["y"] = df_long["y"].astype(str).str.replace(",", ".").astype(float)

# Filtragem para o período solicitado no exercício
df_sf = df_long.query("ds >= '2004-01-01'")[["unique_id", "ds", "y"]].sort_values("ds").reset_index(drop=True)

# Geração de previsões in-sample e resíduos
previsao = sf.forecast(df=df_sf, h=24, level=[80, 95], fitted=True)
df_res = sf.forecast_fitted_values()
df_res['residuo'] = df_res['y'] - df_res['AutoARIMA']
residuos = df_res['residuo'].dropna()

# Plots de diagnóstico
fig, axes = plt.subplots(1, 3, figsize=(18, 4))
axes[0].plot(df_res['ds'], df_res['residuo'])
axes[1].hist(residuos, bins=20, edgecolor='black')
plot_acf(residuos, ax=axes[2], lags=36)
plt.show()

8. Discussão dos Resultados¶

A série original apresentou tendência e sazonalidade, exigindo uma diferenciação regular para atingir a estacionariedade.¶

O diagnóstico dos resíduos não indicou autocorrelação significativa, validando o modelo como ruído branco.¶

A previsão capturou o comportamento sazonal histórico, apresentando intervalos de confiança consistentes com a incerteza do horizonte temporal.¶
